cmake_minimum_required(VERSION 3.14)
project(freetoken-consumer LANGUAGES C CXX)
find_package(freetoken CONFIG REQUIRED)
add_executable(freetoken-consumer ../smoke.cpp)
target_link_libraries(freetoken-consumer PRIVATE freetoken::runtime)
if (MSVC)
    target_compile_options(freetoken-consumer PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME freetoken-consumer COMMAND freetoken-consumer)
