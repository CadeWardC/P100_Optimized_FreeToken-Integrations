cmake_minimum_required(VERSION 3.24)
project(FreeTokenDesktop VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
option(FT_BUILD_FLTK "Build the FLTK desktop application" ${WIN32})
option(FT_BUILD_SLINT "Build the Slint desktop application (requires Rust)" OFF)
# Everything links statically into self-contained executables (per the memory and
# distribution goals): no runtime DLLs beside the exes, no import-libary races.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FLTK_MSVC_RUNTIME_DLL OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FORMS OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FLUID OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_FLTK_OPTIONS OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_GL OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(FT_BUILD_FLTK)
FetchContent_Declare(fltk GIT_REPOSITORY https://github.com/fltk/fltk.git
    GIT_TAG 0d28470949323dedff75444d0011f0b5d38064f1)
FetchContent_MakeAvailable(fltk)
endif()
find_package(Threads REQUIRED)
set(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../llama.cpp/vendor")
add_library(desktop-core STATIC core.cpp configuration.cpp platform.cpp library.cpp "${VENDOR}/cpp-httplib/httplib.cpp")
target_include_directories(desktop-core PUBLIC "${VENDOR}" "${VENDOR}/cpp-httplib" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(desktop-core PUBLIC Threads::Threads)
if(WIN32)
    target_compile_definitions(desktop-core PUBLIC _WIN32_WINNT=0x0A00 NOMINMAX)
    target_link_libraries(desktop-core PUBLIC ws2_32 crypt32 psapi)
endif()
if(MSVC)
    add_compile_options(/utf-8)
endif()
if(FT_BUILD_FLTK)
add_executable(freetoken-desktop WIN32 main.cpp)
set_target_properties(freetoken-desktop PROPERTIES OUTPUT_NAME FreeP100)
target_link_libraries(freetoken-desktop PRIVATE desktop-core fltk::fltk fltk::images)
if(WIN32)
    target_link_libraries(freetoken-desktop PRIVATE dwmapi)
endif()
if(MSVC)
    target_link_options(freetoken-desktop PRIVATE /ENTRY:mainCRTStartup)
endif()
install(TARGETS freetoken-desktop RUNTIME DESTINATION .)
endif()

enable_testing()
add_executable(desktop-tests tests.cpp)
target_link_libraries(desktop-tests PRIVATE desktop-core)
add_test(NAME desktop-core COMMAND desktop-tests)
set_tests_properties(desktop-core PROPERTIES TIMEOUT 30)
if(WIN32 AND FT_BUILD_FLTK)
    add_test(NAME desktop-ui-stream COMMAND freetoken-desktop --integration-test $<TARGET_FILE:desktop-tests> "${CMAKE_CURRENT_BINARY_DIR}/integration-data")
    set_tests_properties(desktop-ui-stream PROPERTIES TIMEOUT 40)
endif()

# Native Slint UI: same engine core, in-process, no browser.
if(FT_BUILD_SLINT)
set(SLINT_STYLE fluent-dark CACHE STRING "Slint widget style")
# Memory-first: CPU software renderer only — zero VRAM, no GPU/OpenGL stack in the GUI process.
set(SLINT_FEATURE_RENDERER_SOFTWARE ON CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_FEMTOVG OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_RENDERER_SKIA OFF CACHE BOOL "" FORCE)
set(SLINT_FEATURE_BACKEND_QT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(Slint GIT_REPOSITORY https://github.com/slint-ui/slint.git
    GIT_TAG release/1 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(Slint)
add_executable(freetoken-slint native/native.cpp)
set_target_properties(freetoken-slint PROPERTIES OUTPUT_NAME llamacpp-p100-native CXX_STANDARD 20)
target_link_libraries(freetoken-slint PRIVATE desktop-core Slint::Slint)
if(WIN32)
    # winit's static-link requirements (Slint docs: system libraries must be listed
    # explicitly when linking the Slint runtime statically)
    target_link_libraries(freetoken-slint PRIVATE imm32 dwmapi shell32 ole32 uuid
        winmm comdlg32 advapi32 user32 gdi32 winspool)
endif()
slint_target_sources(freetoken-slint native/app.slint)

install(TARGETS freetoken-slint RUNTIME DESTINATION .)
endif()

# Headless control-plane host: serves the web UI and manages llama-server. No FLTK.
add_executable(freetoken-host host.cpp control.cpp)
set_target_properties(freetoken-host PROPERTIES OUTPUT_NAME llamacpp-p100)
target_link_libraries(freetoken-host PRIVATE desktop-core)
if(WIN32)
    target_link_libraries(freetoken-host PRIVATE ole32 shell32 uuid)
endif()
add_custom_command(TARGET freetoken-host POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/ui" "$<TARGET_FILE_DIR:freetoken-host>/ui")
install(TARGETS freetoken-host RUNTIME DESTINATION .)

install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/ui" DESTINATION .)
